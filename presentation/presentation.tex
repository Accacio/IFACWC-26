\section{Background}

\begin{frame}{Background}{Constrained Zonotopes \footnote{\articlesymbol\ \cite{scott2016constrained}}}
  Given ${G\in \R^{n\times m}}$, ${\vec{c}\in \R^n}$, ${A\in\R^{c\times m}}$ and ${\vec{b}\in\R^{c}}$
  \begin{equation*}
    \set{CZ}= \langle G,\vec{c},A,\vec{b}\rangle \onslide<+(1)->{= \{G \vec{\xi} + \vec{c} \; \mid \; \norm{\vec{\xi}}_\infty \leq 1\onslide<+(1)->{, A\vec{\xi} = \vec{b}}\}}  \subset \mathbb{R}^n.
  \end{equation*}
  \pause
  \vspace{-.5cm}
  \begin{align*}
    R\set{Z}_a=&\langle RG_a,R\vec{c}_a, A_a,\vec{b}_a\rangle\\
    \set{Z}_a + \set{Z}_b=&\left\langle
                            \left[\begin{smallmatrix}
                              G_a&
                                   G_b
                            \end{smallmatrix}\right] ,\vec{c}_a+\vec{c}_b, \left[\begin{smallmatrix}
                              A_a& 0\\
                              0  & A_b
                            \end{smallmatrix}\right],
                            \left[\begin{smallmatrix}
                              \vec{b}_a\\
                              \vec{b}_b
                            \end{smallmatrix}\right]
                            \right\rangle\tikzmark{plus_node}\\
    \alert<+(1)>{\set{Z}_a\cap_R \set{Z}_c}=&\left\langle
                                               \left[\begin{smallmatrix}
                                                 G_a&
                                                      0
                                               \end{smallmatrix}\right] ,\vec{c}_a, \left[\begin{smallmatrix}
                                                 A_a & 0\\
                                                 0   & A_c\\
                                                 RG_a&  -G_c
                                               \end{smallmatrix}\right],
                                               \left[\begin{smallmatrix}
                                                 \vec{b}_a\\
                                                 \vec{b}_b\\
                                                 \vec{c}_c-R\vec{c}_a
                                               \end{smallmatrix}\right]
                                               \right\rangle\tikzmark{cap_node}
  \end{align*}

\begin{frame}{Background}{Inclusion functions}

Given a function $\vec{y} = \vec{h}(\vec{x})$,
a set-valued linear function
$\hat{\vec{y}} = \vec{y}_0 + H \vec{x} + \epsilon$
is a \linearinclusionlong\ of $\vec{y}$ if
$$\vec{y} \in \vec{y}_0 + H \vec{x} + [\set{\epsilon}], \forall \vec{x} \in \set{X}$$

The set-valued solution $[\hat{\vec{y}}]$ always contains the nonlinear solution $\vec{y}$.

\vfill
\visible<+(1)->{aka Taylor Models when $[\set{\epsilon}]$ calculated using Lagrange remainders}
\end{frame}

\section{Problem Statement}

\begin{frame}{Problem}{State Estimation}
  \begin{block}{}
    Given a system's state and observation dynamics,\\ compute regions ${\set{X}(t) \subset \mathbb{R}^n}$, such that $\vec{x}(t)\in\set{X}(t)$, $\forall t\geq0$.
  \end{block}
\end{frame}

\begin{frame}{System specificaion}

  \vfill
  \begin{block}{State Dynamics}
    \begin{equation*}
      \vec{x}(t_{k}) = \vec{f}(\vec{x}(t_{k-1}),\vec{u}(t_{k-1})) + \alert<+(1)->{\vec{q}(t_{k-1})}
      \label{eq:setfilter_plant_static}
    \end{equation*}
  \end{block}

  % \pause
  \vfill
  \begin{block}{Uncertain observations}
    $\vec{y}(t)\in\R^{m(t)}$

      $
      \begin{bmatrix}
        \vec{y}(t) \\ \hat{t}_y(t)
      \end{bmatrix} = \begin{bmatrix}
          \vec{h}(\vec{x}(t), \vec{u}(t), t) + \alert<+(1)->{\vec{w}(t)} \\
          t + \alert<.(1)->{w_t(t)}
        \end{bmatrix}%\\
      \to$\onslide<+(1)->{\footnote{\onslide<.(1)->{\articlesymbol\ \cite{scott2016constrained}}}}$\onslide<+(1)->{ \overbrace{
        \begin{bmatrix}
          {y}_i(t) \\
          \hat{t}_y(t)
        \end{bmatrix}
        =
        \begin{bmatrix}
          h_i(\vec{x}(t), \vec{u}(t), t) &+& w_{i}(t) \\
          t &+& w_t(t)
        \end{bmatrix}}^{\text{for each of the } m \text{ observations}}}
$
  \end{block}
  
\end{frame}

\begin{frame}{Conditional propagation}

\begin{align*}
  \vec{x}(t_{k+1}) = \vec{f}\left(\vec{x}(t_{k}),\vec{u}(t_{k}), t_{k+1}, t_{k}\right) + \vec{q}(t_{k})
\end{align*}
\small linearized around ${\bar{\vec{o}}(t_{k})=(\bar{\vec{x}}(t_{k}),\bar{\vec{u}}(t_{k}))}$,\\ where ${\bar{\vec{x}} (t_{k}) = \imid{[\post{\set{Z}}_{\vec{x}} (t_{k-1})]}}$ and %\\
  ${\bar{\vec{u}} (t_{k}) = \imid{[\post{\set{Z}}_{\vec{u}} (t_{k-1})]}}$.

\normalsize
\begin{align*}
  \vec{x}(t_{k+1}) = F_{\vec{x}}(t_{k}) \vec{x}(t_{k})
  + F_{\vec{u}}(t_{k}) \vec{u}(t_{k})
  + \vec{\epsilon}_f(t_{k})
  + \vec{c}_f(t_{k}),
  + \vec{q}(t_{k})
\end{align*}
\small where  ${\vec{c}_f(t_{k}) =
  \vec{f}(\bar{\vec{o}}(t_{k}), t_{k-1}, t_{k})
  - F_{\vec{x}}(t_{k}) \bar{\vec{x}}(t_{k})
  - F_{\vec{u}}(t_{k}) \bar{\vec{u}}(t_{k})}$, ${F_{\vec{x}}(t_k) =  \vec{f}_{\vec{x}}(\bar{\vec{o}})}$, ${F_{\vec{u}}(t_k) =  \vec{f}_{\vec{u}}(\bar{\vec{o}})}$,
 ${\vec{\epsilon}_{\vec{f}}(t_k) \in [\vec{\epsilon}_{\vec{f}}(t_k)]}$ is the lumped linearization error,
\normalsize

  \pre{\set{Z}}_{\vec{x}} (t_k) =
